cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(spatial_disorientation C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(spatial_disorientation src/main.cpp src/usb_descriptors.c)
target_include_directories(spatial_disorientation PRIVATE vendor/ComputerCard src)
target_compile_options(spatial_disorientation PRIVATE -Wdouble-promotion -Wfloat-conversion -Wall -Wextra)
target_compile_definitions(spatial_disorientation PRIVATE PICO_XOSC_STARTUP_DELAY_MULTIPLIER=64)
target_link_options(spatial_disorientation PRIVATE -Wl,--print-memory-usage)
target_link_libraries(spatial_disorientation pico_unique_id pico_stdlib hardware_dma hardware_i2c hardware_pwm hardware_adc hardware_spi hardware_flash pico_multicore tinyusb_device tinyusb_board)
pico_enable_stdio_usb(spatial_disorientation 0)
pico_enable_stdio_uart(spatial_disorientation 0)
pico_set_binary_type(spatial_disorientation copy_to_ram)
pico_add_extra_outputs(spatial_disorientation)
